public class LogCaptureService extends Service {
    private static final String TAG = "LogCaptureService";
    private boolean isRunning = false;
    private Process logcatProcess;
    private String logFilePath;

    @Override
    public void onCreate() {
        super.onCreate();
        logFilePath = getExternalFilesDir(null) + "/app_logs.txt";
    }

    @Override
    public int onStartCommand(Intent intent, int flags, int startId) {
        if (!isRunning) {
            isRunning = true;
            startLogCapture();
            Toast.makeText(this, "日志抓取已启动", Toast.LENGTH_SHORT).show();
        }
        return START_STICKY;
    }

    private void startLogCapture() {
        new Thread(() -> {
            try {
                // 清除旧日志
                Runtime.getRuntime().exec("logcat -c");
                
                // 抓取日志并写入文件
                logcatProcess = Runtime.getRuntime().exec("logcat -v time -f " + logFilePath);
                
                // 如果你想限制日志大小，可以添加过滤器
                // logcatProcess = Runtime.getRuntime().exec("logcat -v time *:D -f " + logFilePath);
                
                logcatProcess.waitFor();
            } catch (IOException | InterruptedException e) {
                Log.e(TAG, "日志抓取错误: " + e.getMessage());
            }
        }).start();
    }

    @Override
    public void onDestroy() {
        super.onDestroy();
        if (logcatProcess != null) {
            logcatProcess.destroy();
        }
        isRunning = false;
        Toast.makeText(this, "日志抓取已停止", Toast.LENGTH_SHORT).show();
    }

    @Nullable
    @Override
    public IBinder onBind(Intent intent) {
        return null;
    }
}
